% Caja 3D de nodo UML: \cajaTresDProd{capa}{id}{lista fit}{relleno}{lateral}{profundidad}
\newcommand{\cajaTresDProd}[6]{%
  \begin{pgfonlayer}{#1}
    \node[draw=black, line width=0.7pt, fill=#4, inner sep=7pt, fit={#3}] (#2) {};
    \fill[#5] (#2.north west) -- ++(#6,#6) -- ($(#2.north east)+(#6,#6)$)
          -- ($(#2.south east)+(#6,#6)$) -- (#2.south east) -- (#2.north east) -- cycle;
    \draw[line width=0.7pt] (#2.north west) -- ++(#6,#6) -- ($(#2.north east)+(#6,#6)$)
          -- ($(#2.south east)+(#6,#6)$) -- (#2.south east);
    \draw[line width=0.7pt] (#2.north east) -- ++(#6,#6);
  \end{pgfonlayer}}

% Ícono de documento (artefacto)
\newcommand{\iconoDocProd}[1]{%
  \begin{scope}[shift={($(#1.north east)+(-0.26,-0.2)$)}, line width=0.4pt]
    \draw[fill=white] (-0.09,-0.12) -- (-0.09,0.12) -- (0.03,0.12) -- (0.09,0.06) -- (0.09,-0.12) -- cycle;
    \draw (0.03,0.12) -- (0.03,0.06) -- (0.09,0.06);
  \end{scope}}

% Ícono de componente UML
\newcommand{\iconoCompProd}[1]{%
  \begin{scope}[shift={($(#1.north east)+(-0.26,-0.2)$)}, line width=0.4pt]
    \draw[fill=white] (-0.06,-0.12) rectangle (0.1,0.12);

    \draw[fill=white] (-0.11,-0.08) rectangle (-0.01,-0.03);
  \end{scope}}
